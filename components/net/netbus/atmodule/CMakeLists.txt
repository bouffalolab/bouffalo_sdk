    set(AT_FACTORY_PARAM_FILE "${CMAKE_CURRENT_SOURCE_DIR}/config/${CHIP}/factory_param.csv")
    file(GLOB_RECURSE AT_FACTORY_PARAM_INPUTS CONFIGURE_DEPENDS
        "${CMAKE_CURRENT_SOURCE_DIR}/config/${CHIP}/*")
    get_filename_component(AT_PRIMARY_PT_FILE "${CONFIG_OVERRIDE_PTFILE}"
        ABSOLUTE BASE_DIR "${SDK_DEMO_PATH}")

    add_custom_command(OUTPUT ${AT_USER_KV_IMAGE}
        COMMAND python3 ${CMAKE_CURRENT_SOURCE_DIR}/tools/gen_user.py
            --sdk "${BL_SDK_BASE}" --primary-table "${AT_PRIMARY_PT_FILE}"
            --factory "${AT_FACTORY_PARAM_FILE}" --output "${AT_USER_KV_IMAGE}"
        DEPENDS tools/gen_user.py "${AT_PRIMARY_PT_FILE}" ${AT_FACTORY_PARAM_INPUTS}
        VERBATIM)
    add_custom_target(at_user_kv_image DEPENDS ${AT_USER_KV_IMAGE})
    add_dependencies(${CURRENT_STATIC_LIBRARY} at_user_kv_image)
    sdk_add_flash_partition_image(PARTITION user_kv FILE ${AT_USER_KV_IMAGE})
