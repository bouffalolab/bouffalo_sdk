set(MACSW_PROFILE "${CONFIG_MACSW_PROFILE_NAME}")

sdk_add_subdirectory_ifdef(CONFIG_BLUETOOTH bluetooth)
sdk_add_include_directories(include)
if(CONFIG_MACSW_SELECT STREQUAL "bridge" AND NOT CONFIG_WIFI_WDS)
    message(FATAL_ERROR "MACSW_SELECT=bridge requires CONFIG_WIFI_WDS=y. Three-address mode is runtime enable=0, not CONFIG_WIFI_WDS=n.")
endif()
if(CONFIG_WIFI_WDS AND (NOT CONFIG_FHOST OR CONFIG_WL80211))
    # WDS requires FHOST and cannot be built with WL80211. MAT can be linked in the same image.
    message(FATAL_ERROR "WDS requires FHOST and cannot be built with WL80211")
endif()
sdk_add_subdirectory_ifdef(CONFIG_WL80211 wl80211)

if(CONFIG_WIFI6)
    sdk_add_compile_definitions_ifdef(CONFIG_IPV6 -DCFG_IPV6)
    sdk_add_link_options(-L${CMAKE_CURRENT_SOURCE_DIR}/macsw)
    sdk_add_include_directories(macsw/inc)
    if(NOT MACSW_PROFILE)
        message(FATAL_ERROR "CONFIG_WIFI6 requires one MACSW_PROFILE choice")
    endif()
    sdk_add_compile_definitions(-DCONFIG_MACSW_SELECT_INCLUDE=\"macsw_${MACSW_PROFILE}_config.h\")
    # The released MACSW core is compiled per profile (BGA/P2P switch off
    # compile-time features), so prefer the core matching the selected profile
    # when the release archive ships one.
    if(EXISTS ${CMAKE_CURRENT_SOURCE_DIR}/macsw/lib/libmacsw_${CHIP}_${MACSW_PROFILE}.a)
    sdk_add_static_library(macsw/lib/libmacsw_${CHIP}_${MACSW_PROFILE}.a)
    elseif(EXISTS ${CMAKE_CURRENT_SOURCE_DIR}/macsw/lib/libmacsw_${CHIP}.a)
    sdk_add_static_library(macsw/lib/libmacsw_${CHIP}.a)
    else()
    sdk_add_static_library(${CMAKE_BINARY_DIR}/build_macsw/libmacsw_${CHIP}.a)
    endif()
    if(EXISTS ${CMAKE_CURRENT_SOURCE_DIR}/macsw/lib/libmacsw_config_${CHIP}_${MACSW_PROFILE}.a)
    sdk_add_static_library(macsw/lib/libmacsw_config_${CHIP}_${MACSW_PROFILE}.a)
    else()
    sdk_add_static_library(${CMAKE_BINARY_DIR}/build_macsw/libmacsw_config_${CHIP}_${MACSW_PROFILE}.a)
    endif()
endif()

sdk_add_compile_definitions_ifdef(CONFIG_TIMESLICE_STATS_DEBUG -DMACSW_TIMESLICE_STATS_DEBUG=1)

if(CONFIG_FHOST)
    sdk_add_link_options(-L${CMAKE_CURRENT_SOURCE_DIR}/wifi6)
    sdk_add_include_directories(wifi6/fhost/include)
    if(EXISTS ${CMAKE_CURRENT_SOURCE_DIR}/wifi6/fhost/lib/libfhost_${CHIP}_${MACSW_PROFILE}.a)
    sdk_add_static_library(wifi6/fhost/lib/libfhost_${CHIP}_${MACSW_PROFILE}.a)
    else()
    sdk_add_static_library(${CMAKE_BINARY_DIR}/build_fhost/libfhost_${CHIP}_${MACSW_PROFILE}.a)
    endif()
    sdk_add_subdirectory_ifexists(wifi6)
endif()
sdk_add_subdirectory_ifdef(CONFIG_BL_WPA_SUPPLICANT bl_wpa_supplicant)
sdk_add_subdirectory_ifdef(CONFIG_LPAPP lpfw_private)
if(CONFIG_LMAC154)
    sdk_add_include_directories(lmac154/lmac154/include)
    if(EXISTS ${CMAKE_CURRENT_SOURCE_DIR}/lmac154/lmac154/lib/liblmac154_${CHIP}.a)
      sdk_add_static_library(lmac154/lmac154/lib/liblmac154_${CHIP}.a)
    else()
      sdk_add_include_directories(lmac154/lmac154/src)
      sdk_add_static_library(${CMAKE_BINARY_DIR}/build_lmac154/liblmac154_${CHIP}.a)
    endif()
endif()

if(CONFIG_THREAD)
  sdk_add_subdirectory_ifexists(thread)

  if(CONFIG_OTBR)
    sdk_add_include_directories(thread/openthread_br/include)
    if(EXISTS ${CMAKE_CURRENT_SOURCE_DIR}/thread/openthread_br/lib/libopenthread_br_${CHIP}.a)
      sdk_add_static_library(thread/openthread_br/lib/libopenthread_br_${CHIP}.a)
    else()
      sdk_add_static_library(${CMAKE_BINARY_DIR}/build_openthread_br/libopenthread_br_${CHIP}.a)
    endif()
  endif()
endif()

sdk_add_subdirectory_ifdef(CONFIG_MATTER matter/mfd)
sdk_add_subdirectory_ifdef(CONFIG_SUBG bl3141_drv)
if(CONFIG_ZIGBEE)
  sdk_add_subdirectory_ifexists(zigbee)
endif()
sdk_add_subdirectory_ifdef(CONFIG_WLAN wlan/wlan_mac)
sdk_add_subdirectory_ifdef(CONFIG_WIFI4 wifi4)

if(CONFIG_WIFI6)
sdk_generate_library(macsw_plat)
sdk_library_add_sources(macsw_plat.c)
endif()

# Wi-Fi services follow the configured backend, independently of app sources.
if(CONFIG_WIFI4 OR (CONFIG_WIFI6 AND (CONFIG_FHOST OR CONFIG_WL80211)))
    if(NOT CONFIG_FREERTOS OR NOT CONFIG_RF OR NOT CONFIG_LWIP)
        message(FATAL_ERROR "Wi-Fi startup requires FreeRTOS, RF and lwIP")
    endif()
    if(CONFIG_FHOST AND CONFIG_WL80211)
        message(FATAL_ERROR "Select one Wi-Fi host backend: FHOST or WL80211")
    endif()
    if(CONFIG_WIFI4 AND CONFIG_WIFI6)
        message(FATAL_ERROR "Select one Wi-Fi stack: WIFI4 or WIFI6")
    endif()
endif()
