get_cmake_property(_bflb_config_vars VARIABLES)
foreach(_bflb_config_var IN LISTS _bflb_config_vars)
    if(_bflb_config_var MATCHES "^CONFIG_EXTENSION_[A-Za-z0-9_]+$")
        if(${_bflb_config_var})
            string(REGEX REPLACE "^CONFIG_EXTENSION_" "" _bflb_component_name "${_bflb_config_var}")
            string(TOLOWER "${_bflb_component_name}" _bflb_component_name)
            set(_bflb_component_dir "${BL_SDK_BASE}/extensions/${_bflb_component_name}")
            if(EXISTS "${_bflb_component_dir}/CMakeLists.txt")
                add_subdirectory("${_bflb_component_dir}"
                    "${CMAKE_CURRENT_BINARY_DIR}/extensions/${_bflb_component_name}")
            endif()
        endif()
    endif()
endforeach()
